AI Toolkit

In-editor AI: Ask AI from the toolbar, a docked AI Chat panel, an inline preview of every suggestion, and a persistent AI Review drawer. Runs in built-in demo mode on this page - no API key required.

Launch-ready product update

RichTextEditor now supports in-editor AI actions that stay close to the document instead of opening a disconnected utility panel.

Select a paragraph and click Ask AI in the toolbar to try proofread, rewrite, shorten, expand, summarize, or translate. Each suggestion previews inline - nothing commits until you accept.

  • Inline AI preview with accept / reject controls
  • Docked AI Chat for multi-turn editing help
  • AI Review drawer for pending, accepted, and rejected suggestions

<script>
	var aiEditor = new RichTextEditor("#div_ai_editor", {
		toolbar: "full",
		height: "520px",
		showFloatTextToolBar: true,
		showFloatParagraph: true,
		aiToolkitPersistenceKey: "demo-ai-toolkit"
	});

	function aiDemoSelectFirstParagraph() {
		var doc = aiEditor.getDocument ? aiEditor.getDocument() : document;
		var p = doc.querySelector("p");
		if (!p || !p.firstChild) { aiEditor.focus(); return; }
		var sel = doc.getSelection ? doc.getSelection() : window.getSelection();
		var range = doc.createRange();
		range.setStart(p.firstChild, 0);
		range.setEnd(p.firstChild, p.textContent.length);
		sel.removeAllRanges();
		sel.addRange(range);
		aiEditor.focus();
	}

	function aiDemoOpenDialog() { aiEditor.aiToolkit.openDialog({ presetMode: "rewrite" }); }
	function aiDemoProofread() { aiDemoSelectFirstParagraph(); aiEditor.aiToolkit.runQuickAction("proofread"); }
	function aiDemoSummarize() { aiEditor.aiToolkit.runQuickAction("summarize"); }
	function aiDemoChat() { aiEditor.aiToolkit.openChatPanel({ focusComposer: true }); }
	function aiDemoReview() { aiEditor.aiToolkit.openReviewPanel({ focusPanel: true }); }
	function aiDemoClear() {
		try { localStorage.removeItem("RTE.AIToolkit.demo-ai-toolkit"); } catch (e) {}
		if (aiEditor.aiToolkit && aiEditor.aiToolkit.clearSuggestions) aiEditor.aiToolkit.clearSuggestions();
		alert("Local AI suggestions cleared.");
	}
</script>

Wire your own backend

The editor has no built-in provider. Pass an aiToolkitResolver function and route through your own backend - keys stay server-side. 

var editor = new RichTextEditor("#editor", {
    toolbar: "full",
    aiToolkitResolver: async function (request) {
        var reply = await fetch("/api/ai", {
            method: "POST",
            body: JSON.stringify({
                mode: request.mode, text: request.source, language: request.language
            })
        }).then(function (r) { return r.json(); });

        return {
            result: reply.text,
            reason: reply.explanation,
            operations: [{ type: "preview-suggestion", text: reply.text }]
        };
    }
});
